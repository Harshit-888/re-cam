Image Processing & Filter Mathematical Models

In base.apk, filters are calculated using color matrix transformations and OpenCV contrast stretching. In our rebuilt Kotlin codebase (util/ImageProcessingUtils.kt), these are implemented via hardware-accelerated Android ColorMatrix matrices:

ImageProcessingUtils.kt — Filter ColorMatrix Definitions 
// 1. MAGIC COLOR (Doc Level 1): Boosts saturation by 1.3x and contrast by 1.4x
val MagicColorMatrix = ColorMatrix(floatArrayOf(
    1.4f, 0f,   0f,   0f, -30f,
    0f,   1.4f, 0f,   0f, -30f,
    0f,   0f,   1.4f, 0f, -30f,
    0f,   0f,   0f,   1f,   0f
))

// 2. CLEAR B&W (Doc Level 2): Extreme luminance contrast for crisp document text
val ClearBwMatrix = ColorMatrix(floatArrayOf(
    2.2f, 2.2f, 2.2f, 0f, -280f,
    2.2f, 2.2f, 2.2f, 0f, -280f,
    2.2f, 2.2f, 2.2f, 0f, -280f,
    0f,   0f,   0f,   1f,    0f
))

// 3. GRAYSCALE: Standard ITU-R BT.601 perceptual luminance weights
val GrayscaleMatrix = ColorMatrix().apply { setSaturation(0f) }

// 4. LIGHTEN: Lifts dark shadows while preserving high-contrast text
val LightenMatrix = ColorMatrix(floatArrayOf(
    1.1f, 0f,   0f,   0f,  35f,
    0f,   1.1f, 0f,   0f,  35f,
    0f,   0f,   1.1f, 0f,  35f,
    0f,   0f,   0f,   1f,   0f
))

Domain Data Models & Schemas

model/Models.kt — Core Domain Entities 
data class Document(
    val id: String = UUID.randomUUID().toString(),
    val title: String,
    val createdAt: Long = System.currentTimeMillis(),
    val folderId: String? = null,
    val pages: List<ScannedPage> = emptyList(),
    val isDeleted: Boolean = false,
    val deletedAt: Long? = null,
    val tags: List<String> = emptyList()
)

data class ScannedPage(
    val id: String = UUID.randomUUID().toString(),
    val documentId: String,
    val pageIndex: Int,
    val imageUri: String,
    val filter: FilterType = FilterType.ORIGINAL,
    val rotation: Int = 0,
    val cropCorners: QuadCorners? = null,
    val signatureUri: String? = null,
    val watermarkText: String? = null,
    val ocrText: String? = null
)

data class QuadCorners(
    val topLeft: Offset,
    val topRight: Offset,
    val bottomRight: Offset,
    val bottomLeft: Offset
)

enum class FilterType {
    ORIGINAL, MAGIC_COLOR, CLEAR_BW, GRAYSCALE, LIGHTEN
}
